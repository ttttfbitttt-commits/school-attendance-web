ّ
نظــرًا لمــا تمًثّلــه الخطــط الدراســية مــن ركيــزة تنظيميــة للعمليــة التعليميــة، بوصفهــا األداة التــي تُحــدَد فــي ضوئهــا المواد الدراســية، وعــدد الحصــص، وزمــن التعلم؛
ّ
بمــا يضمــن تحقيــق التــوازن بيــن مكونــات التعلــم فــي مختلــف المراحــل، والمرِّجــع الــذي تُترِّجــم مــن خاللــه السياســات التعليميــة إىل ممارســات منهًجيــة موِّجِهــة
للتخطيــط والتنظيــم والتنفيــذ.
وانطال ًقــا مــن كونهــا ِّجــز ًًءا أصيــل ً ا مــن المنهــج وأحــد الممكنــات الرئيســة لتحســين نواتــج التعل ّ ــم، أوىل المركــز الوطنــي للمناهــج عنايــ ًة خاصــة بتطويــر الخطــط
الدراســية، وفــق منهًجيــة تســتند إىل إشــراك أصحــاب العالقــة، وتحليــل واقــع التطبيــق لإلصــدار الخامــس، وتعزيــز كفــاًءة اســتًثمار الزمــن التعليمــي خالل اليوم الدراســي.
وفــي هــذا الســياق، يأتــي اإلصــدار الســادس مــن دليــل الخطــط الدراســية بعــد سلســلة مــن اإلِّجــراًءات التطويريــة؛ دع ًمــا لرفــع ِّجــودة التعلــم، وتعزيــز مواًءمتــه مــع
التوِّجهــات الوطنيــة ومســتهدفات رؤيــة المملكــة 0302.
ومــن المؤ ّمــل أن يشــك ّ ل هــذا اإلصــدار مرِّج ًعــا فاعــل ً ا لكافــة المعنييــن بالعمليــة التعليميــة، مــن القيــادات التربويــة والمعلميــن وأوليــاًء األمــور والــطالب، إىل
ّ
ِّجانــب األكاديمييــن والباحًثيــن؛ نظــرًا لمــا يمًثّلــه التطويــر المســتمر للخطــط الدراســية مــن عنصــر محــوري فــي تحســين نواتــج التعلــم، وتعزيــز كفــاًءة النظــام التعليمــي
في المملكة.

أداة لتحديــد الــوزن النســبي المناســب للمــواد الدراســية واألنشــطة الطالبيــة ضمــن مًجــاالت التعلــم المســتهدفة ووفــق مراحــل وأنــواع التعليــم المختلفــة خالل
الخطط الدراسية
العام الدراســي.
أداة تنظيميــة تُعتمــد لفتــرة زمنيــة محــددة اســتًجاب ًة لظــروف أو متغيــرات قائمــة (تعليميــة أو تنظيميــة أو تشــغيلية)، وتهــدف إىل إتاحــة نمــاذج تعليميــة بديلــة أو
الخطط الدراسية المؤقتة
مطــ ّورة تتــواًءم مــع طبيعــة تلــك الظــروف، وتســهم فــي دعــم المدرســة لتحقيــق أهدافهــا.
ِّجميــع المــدارس للمراحــل االبتدائيــة والمتوســطة والًثانويــة، ومــدارس تحفيــظ القــرآن الكريــم، ومــدارس الطفولــة المبكــرة، ومــدارس التعليــم المســتمر،
التعليم العام
والمــدارس الســعودية فــي الخــارج، ومــا فــي حكمهــا مــن مؤسســات التعليــم العــام التــي تــدار مــن قبــل وزارة التعليــم وقطاعاتهــا.
المرحلــة التــي تُعنــى برعايــة األطفــال الذيــن تتــراوح أعمارهــم مــن المــيالد وحتــى 63 شــهراً، وتهــدف إىل توفيــر الخدمــات فــي بيئــة تراعــي احتيــاج الطفــل، مــع
حضانة األطفال
توفيــر فــرص التعلّــم المبكــر، والتفــاعالت المتًجاوبــة، والداعمــة عاطفيــاً، والمحفــزة نمائيًــا.
المرحلــة التمهيديــة التــي تُعنــى بتربيــة الطفــل ورعايتــه وتعليمــه (مــن عمــر 3 ســنوات وحتــى 6 ســنوات) وتســبق دخولــه المرحلــة االبتدائيــة وتهيئــه لاللتحــاق
رياض األطفال بهــا، وتهــدف إىل تقديــم خبــرات تعلــم مبكــرة مالئمــة لخصائــص نمــو األطفــال؛ لتُمكّنــه مــن النمــو الســليم المتــوازن والمتكامــل فــي ِّجميــع الًجوانــب، مــن خالل
برامــج تشــمل: اللعــب، والترفيــه، والتعليــم، وفــق اهتمامــات األطفــال ودوافــع نموهــم.
المرحلــة التــي تُعنــى بتربيــة الطالــب وتعليمــه فــي الفئــة العمريــة مــن 6 ســنوات تقريبًــا، وتركــز عىل إكســابه المعــارف والمهــارات، إضافــة إىل تنميــة عــدد مــن
االبتدائية الًجوانــب المعرفيــة والشــخصية والقيميــة. وهــي عىل قســمين: الصفــوف األوليــة وتشــمل (األول والًثانــي والًثالــث)، الصفــوف العليــا وتشــمل (الرابــع والخامــس
والســادس).
المرحلــة التعليميــة التــي تلــي المرحلــة االبتدائيــة وتمهــد للمرحلــة الًثانويــة، وتركــز عىل تعزيــز مــا اكتســبه الطالــب مــن معــارف ومهــارات وقيــم، وتوســيع مداركــه
المتوسطة
فــي المًجــاالت المتنوعــة.

المرحلــة التعليميــة التــي تمًثــل نهايــة التعليــم العــام، وتعــد الطالــب لتلبيــة احتياِّجــات ســوق العمــل ومهــن المســتقبل، كمــا تؤهلــه لمواصلــة تعليمــه الًجامعــي
الًثانوية العامة
بمــا يتناســب مــع متطلبــات التنميــة، وتتكــون مــن ســنة مشــتركة وبرنامــج يتضمــن خمــس مســارات نوعيــة متخصصــة.
المدارس التي توفر الخدمات التعليمية لألطفال (من عمر شهر وحتى الصف الًثالث االبتدائي)، وتشمل: حضانات األطفال، ورياض األطفال، والصفوف األولية
مدارس الطفولة المبكرة
(بنين، وبنات).
مدارس تحفيظ القرآن الكريم المدارس التي تُعنى بتحفيظ القرآن الكريم، إضافة إىل تدريس ِّجميع المواد التعليمية التي تُدرّس في التعليم العام في المرحلتين االبتدائية والمتوسطة.
المدارس السعودية في الخارج المدارس التي توفر الخدمات التعليمية ألبناًء السعوديين المبتعًثين والموفدين والعاملين في الخارج.
مدارس التعليم المستمر المدارس التي توفر الخدمات التعليمية لمن تًجاوزوا سن التعليم اإللزامي، ولم يحظوا بفرصة مواصلة تعليمهم في مدارس التعليم العام.
نموذج تعليمي يسعى إىل تخريج طالب مؤهلين علميًا وتربويًا، ويركز عىل تدريس علوم الكتاب والسنة والعلوم الشرعية واللغة العربية، وله عناية خاصة بدراسة
دار الحديث المكية/ المدنية
الحديث النبوي الشريف وأمهات كتب الحديث.
المدارس التي تدرس منه ًًجا مغايرًا لمنهج التعليم مدارس التعليم الخاص الحاصلة عىل ترخيص من وزارة التعليم، لتقديم منهج مغاير للمنهج الحكومي ومعتمد من مؤسسات دولية معترف بها، وتشمل
الحكومي المدارس األهلية والعالمية.
المدارس المطبقة للغة الصينية المنشأة التعليمية التي تقع ضمن عينة التطبيق المختارة من وزارة التعليم، لتدريس مادة اللغة الصينية كإحدى المواد الدراسية األساسية.
المدارس المطبقة لمبادرة الفنون المنشأة التعليمية التي تقع ضمن عينة التطبيق المختارة من وزارة التعليم، لتدريس مواد الًثقافة والفنون (البصرية، الموسيقية، األدائية) كمواد دراسية أساسية.

د
المنشأة التعليمية التي تقدم المواد الدراسية والخطط والبرامج التعليمية المصممة لتلبية االحتياِّجات التعليمية للطالب من ذوي اإلعاقة، وتشمل:
برامج الدمج الكلي، وتُعنى بإلحاق الطالب في فصول التعليم العام، وتقديم منهج التعليم العام مع توفير الخدمات المساندة والدعم التعليمي المناسب.
برامــج الدمــج المكانــي وتعنــي الفصــول الملحقــة بمــدارس التعليــم العــام، ويدمــج فيهــا الطــالب ذوي اإلعاقــة دم ًًجــا مكانيًــا، مــع تلقــي الخدمــات التعليميــة فــي فصــول
خاصــة أو برامــج منفصلــة داخــل المبنــى المدرســي، ويشــاركون أقرانهــم فــي ِّجميــع الفتــرات واألنشــطة والبرامــج.
برامج ومعاهد ومراكز ذوي اإلعاقة
المعاهــد ومراكــز الخدمــات التعليميــة والتأهيليــة التــي تُعنــى بتوفيــر بيئــات تعليميــة تــالًءم احتياِّجــات الطــالب الًجســدية والمعرفيــة والســلوكية، وتقــدم برامــج تعليميــة
فرديــة لتحســين نواتــج تعلمهــم وتكييفهــم فــي المًجتمــع، وتشــمل: اضطــراب طيــف التوحــد (التحديــات النمائيــة)، التربيــة الفكريــة، تعــدد اإلعاقــات.
برامــج التعليــم المســتمر لــذوي اإلعاقــة الفكريــة، وهــو مســار تعليمــي تأسيســي يقــدم كأحــد أنــواع التعليــم المســتمر، وموِّجــه لألفــراد مــن ذوي اإلعاقــة الفكرية البســيطة
ممــن تًجــاوزا ســن 12 عا ًمــا ولــم يســبق لهــم االلتحــاق بــأي برنامــج تعليمــي نظامــي، ويقــدم مناهــج تلبــي االحتياِّجــات الفرديــة والتعليميــة لهم.
المنشأة التعليمية التي تقدم البرامج التعليمية المصممة لتلبية االحتياِّجات التعليمية لطالب التفوق واالبتكار، وتشمل:
المدارس التخصصية وهو مسار تُقدم فيه الخدمات التعليمية للطالب وفق مقاييس ومحددات معتمدة تختص بها كل مدرسة حسب مًجالها.
المدارس المطبقة لبرامج الموهوبين
فصول الموهوبين التي يًجمع فيها الطالب الموهوبين بفصول مستقلة؛ لتلقي التعليم المتمايز والمتنوع، وفق األساليب العلمية في تدريس الموهوبين.
برامــج الرعايــة المدرســية حيــث يســحب الطــالب الموهوبيــن (tuo lluP)، فــي بيئــة محــددة توفــر برامــج نوعيــة متكاملــة وخبــرات تربويــة تتســم بالتنــوع والعمــق العلمــي،
ويقــدم أثنــاًء الــدوام المدرســي الصباحــي للطــالب الموهوبيــن فــي المــدارس التــي يتوافــر فيهــا معلــم/ ة موهوبيــن.
أيام التمدرس عدد األيام الدراسية خالل العام، والمحددة بما ال يقل عن (081) يو ًًما دراسيًا.
الوقــت المخصــص والمحــدد للدراســة بمــا ال يتًجــاوز ســبع ســاعات عىل ِّجميــع المــواد الدراســية والنشــاط والفتــرات الالصفيــة داخــل البيئــة المدرســية، وفــق
اليوم الدراسي
خطــة زمنيــة منظمــة ومحــددة تبــدأ مــن دخــول الــطالب وحتــى انصرافهــم.
المًجــاالت التعليميــة التخصصيــة المحــددة لــكل صــف ومرحلــة، مًثــل: القــرآن الكريــم والدراســات اإلسالميــة، اللغــة العربيــة، الدراســات االِّجتماعيــة،
المواد الدراسية
الرياضيــات،......... وغيرهــا.
